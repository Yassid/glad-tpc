\begin{frame}{Geometry -- why the chamber is hard}
HYDRA Prototype, chamber-local frame:
\begin{itemize}
  \item Pad plane $8.8 \times 25.6$ cm (bending $\times$ beam).
  \item $\vec{B} = (0, 2~\text{T}, 0)$ -- GLAD horizontal dipole. Bending plane $(x, z)$.
  \item Target (LH$_2$) at local $(-6.9, 0, -33.2)$ cm:
        \textbf{6.9 cm outside the pad-plane in $x$.}
\end{itemize}
\vfill
\begin{columns}[c,onlytextwidth]
\column{0.55\textwidth}
\centering
\textbf{Sagitta vs hit noise}\\($p = 800$ MeV/c $\pi^-$, in-pad only)\\[0.4em]
\large
\begin{tabular}{lr}
\toprule

\bottomrule

$p_T = p\sin\theta_y$ since $\vec{B}\,\|\,\hat{y}$.
\end{columns}

\begin{frame}{The lever -- Gluckstern's bound}
For $N$ hits with noise $\sigma_{xy}$ along chord $L$:
\[
  \boxed{\;\frac{\sigma_R}{R} \;=\; \frac{R\,\sigma_{xy}}{L^2}\sqrt{\frac{720}{N}}\;}
\]
\vspace{-0.5em}

\scriptsize
\begin{columns}[t,onlytextwidth]
\column{0.48\textwidth}
\begin{tabular}{@{}ll@{}}
$\sigma_R/R$ & fractional radius resolution (= $\sigma_p/p$ in $B$-uniform) \\
$R$          & track radius; $R = p_T / (0.3 B)$ [cm, MeV/c, T] \\
$\sigma_{xy}$ & hit resolution in the bending $(x,z)$ plane, $\approx 1$ mm \\
\end{tabular}
\column{0.48\textwidth}
\begin{tabular}{@{}ll@{}}
$L$ & chord length (first hit $\to$ last hit) in $(x,z)$ \\
$N$ & number of hits used in the fit \\
720 & integration constant for uniform sampling \\
\end{tabular}
\end{columns}
\vspace{0.3em}

\begin{center}
\large
\begin{tabular}{lcc}
\toprule
$L$                              & Gluckstern floor & measured \\
\midrule
9 cm (in-pad only)               & 44 \%          & 22.6 \% (baseline) \\
\textbf{16 cm (in-pad + vertex)} & \textbf{14 \%} & \textbf{3.1 \%} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.3em}

$\sigma_R \propto R / L^2$. \textbf{Doubling $L$ drops the floor 4$\times$.}
One pseudo-hit at the target $\Rightarrow$ \textbf{3$\times$ resolution gain} in practice.
\end{frame}

\begin{frame}{Pipeline -- key changes vs naive port}
\small
\begin{tabular}{ll}
\toprule
\textbf{Stage} & \textbf{Change} \\
\midrule
sim (Geant4) & \texttt{GTPC\_STEMAX\_CM=0.1} $\Rightarrow{\sim}22\times$ denser MC \\
TripClust    & \texttt{fMinClusters}\ 5 $\to$ \textbf{3} (low-$p$ recovery) \\
Circle seed  & Pratt + Gauss--Newton + \textbf{vertex pseudo-hit} + Huber \\
$\theta_y$   & helix pitch from $y$-vs-$\varphi$ linear fit \\
UKF (OpenKF) & helix-tangent seed dir; \texttt{fMomSigmaFrac}\ 0.10 $\to$ \textbf{0.02} \\
\bottomrule
\end{tabular}
\vspace{0.4em}

\textbf{Vertex pseudo-hit} -- the breakthrough:
\vspace{-0.3em}
\[
  \chi^2 = \sum_{i} r_i^2 + w_v\, r_v^2,
  \quad
  r_v = \sqrt{(x_v - c_x)^2 + (z_v - c_z)^2} - R,
  \quad
  w_v = (\sigma_{xy}/\sigma_v)^2
\]
\vspace{-0.5em}

\scriptsize
\begin{columns}[t,onlytextwidth]
\column{0.48\textwidth}
\begin{tabular}{@{}l p{4.5cm}@{}}
$r_i = d_i - R$ & hit geometric residual; $d_i = \sqrt{(x_i{-}c_x)^2{+}(z_i{-}c_z)^2}$ \\
$r_v$           & same form, at the vertex \\
$(x_v, z_v)$    & vertex in chamber-local; $x_v = -6.9$ cm fixed by geometry \\
$(c_x, c_z), R$ & fitted circle centre + radius (3 free GN params) \\
\end{tabular}
\column{0.48\textwidth}
\begin{tabular}{@{}l p{4.0cm}@{}}
$w_v$        & vertex weight vs one hit \\
$\sigma_{xy} \approx 1$ mm & reco hit resolution in $(x,z)$ \\
$\sigma_v \approx 0.5$ mm  & beam-spot width \\
$\Rightarrow w_v = 4$ & vertex counts as 4 hits in the GN sum \\
\end{tabular}
\end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Seed fit -- four ingredients}
\small
\begin{columns}[t,onlytextwidth]
\column{0.50\textwidth}
\textbf{1. Pratt's algebraic circle fit} \\
Closed-form: cubic in a Lagrange multiplier (Chernov ch.\,5). \\
\textcolor{good}{Unbiased to leading order in $\sigma_{xy}/L$.}
Naive Kasa biases $R$ low by $\sim 40\,\%$ when sagitta $\approx$ hit noise.
\vspace{0.6em}

\textbf{2. Gauss--Newton refinement} \\
Minimises the \emph{geometric} cost
\[
  \chi^2 = \sum_i (d_i - R)^2, \quad d_i = \sqrt{(x_i-c_x)^2 + (z_i-c_z)^2}
\]
2--5 iterations from the Pratt seed.

\column{0.50\textwidth}
\textbf{3. Vertex pseudo-hit} \\
One residual at the known target, weight $w_v = (\sigma_{xy}/\sigma_v)^2 = 4$. \\
Effective chord $9 \to 16$ cm
$\Rightarrow$ \textcolor{good}{\textbf{3$\times$ tighter $R$}} (Gluckstern $1/L^2$).
\vspace{0.6em}

\textbf{4. Huber loss} \\
Per-hit weight in the GN sum:
\[
  w_i(r_i) = \min(1,\;k/|r_i|), \quad k = 0.10~\text{cm}
\]
Outliers ($\delta$-electrons, mis-clusterings) down-weighted $\propto 1/|r|$
instead of pulling the fit as $r^2$.

\end{columns}
\vspace{0.4em}

\scriptsize
\textit{All four feed the same $(c_x, c_z, R)$ minimisation.}
The Pratt seed makes the GN converge in $\le 5$ steps;
the vertex adds the lever arm;
Huber catches what neither sees.
\end{frame}

\begin{frame}{Open items}
\begin{enumerate}
  \item \textbf{Real-data vertex source}: external beam tracker replaces MC sidecar. $x_v \approx -6.9$ cm fixed by geometry.
  \item \textbf{Below 400 MeV/c}: TripClust short-chord ceiling. 2-cluster fitter or revisit thresholds.
  \item \textbf{Un-instrumented gas} ($\sim 7$ cm target $\to$ chamber): $dE/dx$ + MS not in UKF back-extrap; $<1\,\%$ for MIP $\pi$.
  \item \textbf{Straight-line degeneracy} ($\sim 1\,\%$): clamped at $R = 20$ m. Wants a $\chi^2$ cut or low-curvature flag.
\end{enumerate}
